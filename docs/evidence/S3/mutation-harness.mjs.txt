import fs from 'node:fs/promises';
import os from 'node:os';
import path from 'node:path';
import {validateContent} from '/Users/rexxa/bs-mobile/tools/validate-content.mjs';
const original='/Users/rexxa/bs-mobile/data';
const cases=[];
const change=async(root,file,fn)=>{const p=path.join(root,file);const r=JSON.parse(await fs.readFile(p,'utf8'));fn(r);await fs.writeFile(p,JSON.stringify(r));};
cases.push(['missing tool growth',r=>change(r,'tools/bee_hive.json',x=>delete x.growth)]);
cases.push(['wrong evolution target',r=>change(r,'evolutions/seed_crown.json',x=>x.growthCondition.target='people')]);
cases.push(['duplicate recipe property reorder',async r=>{const a=JSON.parse(await fs.readFile(path.join(r,'evolutions/seed_crown.json'),'utf8'));await change(r,'evolutions/deep_furrow.json',b=>{const id=b.id;Object.assign(b,a,{id});b.growthCondition={minimum:a.growthCondition.minimum,state:a.growthCondition.state,target:a.growthCondition.target};});}]);
cases.push(['skin numeric nested',r=>change(r,'skins/harvest_cloak.json',x=>x.appearance.effects=1)]);
cases.push(['skin hidden stats',r=>change(r,'skins/harvest_cloak.json',x=>x.appearance.damage='10')]);
cases.push(['orphan evolution output',r=>change(r,'evolutions/seed_crown.json',x=>x.result.baseId='core:iron_blade')]);
cases.push(['invalid loop stage',r=>change(r,'tools/bee_hive.json',x=>x.loopLinks[0].stage='fake_stage')]);
cases.push(['self anti synergy',r=>change(r,'tools/bee_hive.json',x=>{x.antiSynergy=[x.id];x.antiSynergyNotes=[{otherId:x.id,reason:'self'}];})]);
cases.push(['canonical test reference',r=>change(r,'heroes/founder.json',x=>x.affinityEstateIds=['test:moor'])]);
cases.push(['test metadata missing reference',r=>change(r,'test/heroes/scout.json',x=>x.affinityEstateIds=['core:missing'])]);
cases.push(['false implemented candidate status',r=>change(r,'tools/bee_hive.json',x=>x.designStatus='s2-runtime')]);
cases.push(['baseline profile removes enemy',r=>change(r,'profiles/s2-baseline.json',x=>x.selection.enemies.pop())]);
cases.push(['baseline profile removes tests',r=>change(r,'profiles/s2-baseline.json',x=>x.testSelection.heroes=[])]);
cases.push(['test tool orphan anti note',r=>change(r,'test/tools/test_spade.json',x=>x.antiSynergyNotes=[{otherId:'core:missing',reason:'broken reference'}])]);
const results=[];
for(const[name,mutate]of cases){const root=await fs.mkdtemp(path.join(os.tmpdir(),'bs-s3-mutation-'));try{await fs.cp(original,root,{recursive:true});await mutate(root);const result=await validateContent(root,{fullPool:true});results.push({name,rejected:!result.valid,errors:result.errors});console.log(JSON.stringify(results.at(-1)));}finally{await fs.rm(root,{recursive:true,force:true});}}
await fs.writeFile('/tmp/bs-s3-mutations.json',JSON.stringify(results,null,2));
