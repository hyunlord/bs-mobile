const {chromium}=require('playwright');
const fs=require('fs');const cp=require('child_process');const path=require('path');
const base='/Users/rexxa/bs-mobile/docs/metrics';const out='/Users/rexxa/bs-mobile/docs/evidence/S4/metrics';fs.mkdirSync(out,{recursive:true});
fs.writeFileSync('/tmp/bs-metrics-browser/s4-baseline.html',cp.execFileSync('git',['show','main:docs/metrics/index.html'],{cwd:'/Users/rexxa/bs-mobile'}));
(async()=>{const browser=await chromium.launch(); const results=[];try{const ref=await browser.newPage({viewport:{width:1280,height:1000}});await ref.goto('file:///tmp/bs-metrics-browser/s4-baseline.html');await ref.screenshot({path:out+'/reference-1280.png'});await ref.close();for(const width of [375,768,1280]){const page=await browser.newPage({viewport:{width,height:1000}});const errors=[];page.on('pageerror',e=>errors.push(String(e)));await page.goto('file://'+base+'/index.html');await page.screenshot({path:out+`/actual-${width}.png`});const record=await page.evaluate(()=>({width:innerWidth,bodyWidth:document.body.scrollWidth,rows:document.querySelectorAll('tbody tr').length,svg:document.querySelectorAll('svg').length,links:[...document.querySelectorAll('a')].map(a=>a.getAttribute('href')),headings:[...document.querySelectorAll('section h2')].map(e=>e.textContent)}));await page.keyboard.press('Tab');record.focus=await page.evaluate(()=>document.activeElement.tagName);for(const [i,h] of record.headings.entries()){if(h.includes('S4')){await page.locator('section').nth(i).scrollIntoViewIfNeeded();await page.screenshot({path:out+`/actual-${width}-series${i}.png`});}}record.errors=errors;record.linksValid=record.links.every(h=>JSON.parse(fs.readFileSync(path.join(base,h))).data.commit);if(record.bodyWidth>width||record.rows!==20||errors.length||!record.linksValid)throw Error(JSON.stringify(record));results.push(record);await page.close();}fs.writeFileSync(out+'/browser.json',JSON.stringify({browser:browser.version(),results},null,2));console.log(JSON.stringify(results.map(({width,bodyWidth,rows,svg,focus,linksValid,errors,headings})=>({width,bodyWidth,rows,svg,focus,linksValid,errors,headings}))));}finally{await browser.close();}})().catch(e=>{console.error(e);process.exitCode=1});
